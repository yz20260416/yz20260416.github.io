#!/usr/bin/env python3
"""Generate city guide pages using the DeepSeek API.

Reads a hardcoded list of 50 cities, calls the DeepSeek chat API to produce
content for each city, renders it into the tokyo.html template, writes the
result to cities/<slug>.html, and regenerates sitemap.xml.
"""

import json
import os
import re
import sys
import time
import urllib.request
import urllib.error
from pathlib import Path

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------

BASE_URL = os.environ.get("SITE_BASE_URL", "https://example.com")
DEEPSEEK_API_KEY = os.environ.get("DEEPSEEK_API_KEY", "")
DEEPSEEK_API_URL = os.environ.get(
    "DEEPSEEK_API_URL", "https://api.deepseek.com/chat/completions"
)
DEEPSEEK_MODEL = os.environ.get("DEEPSEEK_MODEL", "deepseek-chat")

ROOT = Path(__file__).resolve().parent
CITIES_DIR = ROOT / "cities"
TEMPLATE_PATH = CITIES_DIR / "tokyo.html"
SITEMAP_PATH = ROOT / "sitemap.xml"

CITIES = [
    "Tokyo", "Kyoto", "Osaka", "Seoul", "Beijing", "Shanghai", "Bangkok",
    "Singapore", "Bali", "Hanoi",
    "Paris", "London", "Rome", "Barcelona", "Amsterdam", "Vienna", "Prague",
    "Athens", "Lisbon", "Dublin",
    "New York", "Los Angeles", "San Francisco", "Miami", "Toronto", "Vancouver",
    "Mexico City", "Rio de Janeiro", "Buenos Aires", "Lima",
    "Sydney", "Melbourne", "Auckland", "Fiji", "Bora Bora",
    "Cairo", "Cape Town", "Marrakech", "Nairobi", "Zanzibar",
    "Dubai", "Istanbul", "Jerusalem", "Petra", "Maldives",
    "Moscow", "St. Petersburg", "Warsaw", "Budapest", "Copenhagen",
]

# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------


def slugify(name: str) -> str:
    slug = name.lower().strip()
    slug = re.sub(r"[^a-z0-9]+", "-", slug)
    return slug.strip("-")


def call_deepseek(prompt: str, retries: int = 3) -> str:
    """Call the DeepSeek chat API and return the assistant message text."""
    if not DEEPSEEK_API_KEY:
        raise RuntimeError("DEEPSEEK_API_KEY environment variable is not set")

    payload = {
        "model": DEEPSEEK_MODEL,
        "messages": [
            {
                "role": "system",
                "content": (
                    "You are a travel content writer. Respond ONLY with valid "
                    "JSON matching the requested schema. No markdown fences."
                ),
            },
            {"role": "user", "content": prompt},
        ],
        "temperature": 0.7,
        "response_format": {"type": "json_object"},
    }
    data = json.dumps(payload).encode("utf-8")
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}",
    }

    last_err = None
    for attempt in range(1, retries + 1):
        try:
            req = urllib.request.Request(
                DEEPSEEK_API_URL, data=data, headers=headers, method="POST"
            )
            with urllib.request.urlopen(req, timeout=120) as resp:
                body = json.loads(resp.read().decode("utf-8"))
            return body["choices"][0]["message"]["content"]
        except (urllib.error.URLError, KeyError, json.JSONDecodeError) as exc:
            last_err = exc
            print(f"  API error (attempt {attempt}/{retries}): {exc}", file=sys.stderr)
            time.sleep(2 * attempt)
    raise RuntimeError(f"DeepSeek API failed after {retries} attempts: {last_err}")


def build_prompt(city: str) -> str:
    return f"""Write content for a travel guide page titled "Best Time to Visit {city}".
Return a JSON object with exactly these keys:
{{
  "intro": "2-3 sentence overview of the best time to visit {city}.",
  "months": [
    {{"month": "January", "temp_c": "5", "rain_mm": "50", "crowds": "Low", "recommendation": "Good"}},
    ... one entry for each month January through December ...
  ],
  "attractions": [
    {{"name": "Attraction name", "description": "One sentence description."}},
    ... exactly 5 entries ...
  ],
  "practical": {{
    "currency": "...",
    "language": "...",
    "timezone": "...",
    "visa": "..."
  }},
  "faq": [
    {{"question": "...", "answer": "..."}},
    ... exactly 3 entries ...
  ],
  "meta_description": "A unique 150-160 character meta description for this page."
}}
Use realistic values. Keep descriptions concise."""


def render_page(city: str, data: dict) -> str:
    slug = slugify(city)
    title = f"Best Time to Visit {city} - Monthly Weather, Crowds &amp; Tips"
    meta = data.get("meta_description", f"Best time to visit {city}.").replace('"', "&quot;")

    month_rows = "\n".join(
        f"          <tr><td>{m['month']}</td><td>{m['temp_c']}</td>"
        f"<td>{m['rain_mm']}</td><td>{m['crowds']}</td>"
        f"<td>{m['recommendation']}</td></tr>"
        for m in data["months"]
    )

    attraction_items = "\n".join(
        f"      <li><strong>{a['name']}</strong> — {a['description']}</li>"
        for a in data["attractions"]
    )

    p = data["practical"]
    practical_items = (
        f"      <li><strong>Currency:</strong> {p['currency']}</li>\n"
        f"      <li><strong>Language:</strong> {p['language']}</li>\n"
        f"      <li><strong>Time zone:</strong> {p['timezone']}</li>\n"
        f"      <li><strong>Visa:</strong> {p['visa']}</li>"
    )

    faq_items = "\n".join(
        f"      <details>\n        <summary>{f['question']}</summary>\n"
        f"        <p>{f['answer']}</p>\n      </details>"
        for f in data["faq"]
    )

    related = [c for c in CITIES if c != city][:3]
    related_items = "\n".join(
        f'        <li><a href="{slugify(c)}.html">Best time to visit {c}</a></li>'
        for c in related
    )

    return f"""<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>{title}</title>
  <meta name="description" content="{meta}">
  <link rel="canonical" href="{BASE_URL}/cities/{slug}.html">
  <link rel="stylesheet" href="../css/style.css">
</head>
<body>
  <header class="site-header">
    <a class="brand" href="../index.html">Travel Guides</a>
  </header>

  <main>
    <h1>Best Time to Visit {city}</h1>
    <p class="lead">{data['intro']}</p>

    <h2>Monthly Overview</h2>
    <div class="table-wrap">
      <table>
        <thead>
          <tr>
            <th>Month</th>
            <th>Avg Temp (°C)</th>
            <th>Rainfall (mm)</th>
            <th>Crowds</th>
            <th>Recommendation</th>
          </tr>
        </thead>
        <tbody>
{month_rows}
        </tbody>
      </table>
    </div>

    <h2>Top 5 Attractions</h2>
    <ul class="clean">
{attraction_items}
    </ul>

    <h2>Practical Information</h2>
    <ul class="clean">
{practical_items}
    </ul>

    <h2>FAQ</h2>
    <div class="faq">
{faq_items}
    </div>

    <div class="related-links">
      <h2>Related Cities</h2>
      <ul>
{related_items}
      </ul>
    </div>
  </main>

  <footer class="site-footer">
    &copy; 2026 Travel Guides
  </footer>
</body>
</html>
"""


def write_sitemap() -> None:
    """Regenerate sitemap.xml from all HTML files under cities/ plus index."""
    urls = [f"{BASE_URL}/"]
    for html in sorted(CITIES_DIR.glob("*.html")):
        urls.append(f"{BASE_URL}/cities/{html.name}")

    entries = "\n".join(
        f"  <url>\n    <loc>{u}</loc>\n    <changefreq>monthly</changefreq>\n"
        f"    <priority>{'1.0' if u.endswith('/') else '0.8'}</priority>\n  </url>"
        for u in urls
    )
    SITEMAP_PATH.write_text(
        '<?xml version="1.0" encoding="UTF-8"?>\n'
        '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n'
        f"{entries}\n</urlset>\n",
        encoding="utf-8",
    )
    print(f"Wrote sitemap with {len(urls)} URLs -> {SITEMAP_PATH}")


def main() -> None:
    CITIES_DIR.mkdir(parents=True, exist_ok=True)
    if not TEMPLATE_PATH.exists():
        print(f"Template not found: {TEMPLATE_PATH}", file=sys.stderr)
        sys.exit(1)

    for city in CITIES:
        slug = slugify(city)
        out_path = CITIES_DIR / f"{slug}.html"
        print(f"Generating {city} -> {out_path.name}")
        try:
            raw = call_deepseek(build_prompt(city))
            data = json.loads(raw)
            html = render_page(city, data)
            out_path.write_text(html, encoding="utf-8")
        except Exception as exc:  # noqa: BLE001
            print(f"  Failed for {city}: {exc}", file=sys.stderr)

    write_sitemap()


if __name__ == "__main__":
    main()

def slugify(name: str) -> str:
    slug = name.lower().strip()
    slug = re.sub(r"[^a-z0-9]+", "-", slug)
    return slug.strip("-")


def call_deepseek(prompt: str, retries: int = 3) -> str:
    if not DEEPSEEK_API_KEY:
        raise RuntimeError("DEEPSEEK_API_KEY environment variable is not set")

    payload = {
        "model": DEEPSEEK_MODEL,
        "messages": [
            {
                "role": "system",
                "content": (
                    "You are a travel content writer. Respond ONLY with valid "
                    "JSON matching the requested schema. No markdown fences."
                ),
            },
            {"role": "user", "content": prompt},
        ],
        "temperature": 0.7,
        "response_format": {"type": "json_object"},
    }
    data = json.dumps(payload).encode("utf-8")
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {DEEPSEEK_API_KEY}",
    }

    last_err = None
    for attempt in range(1, retries + 1):
        try:
            req = urllib.request.Request(
                DEEPSEEK_API_URL, data=data, headers=headers, method="POST"
            )
            with urllib.request.urlopen(req, timeout=120) as resp:
                body = json.loads(resp.read().decode("utf-8"))
            return body["choices"][0]["message"]["content"]
        except (urllib.error.URLError, KeyError, json.JSONDecodeError) as exc:
            last_err = exc
            print(f"  API error (attempt {attempt}/{retries}): {exc}", file=sys.stderr)
            time.sleep(2 * attempt)
    raise RuntimeError(f"DeepSeek API failed after {retries} attempts: {last_err}")


def render_body(page_type: str, data: dict) -> str:
    parts = [f"    <p class=\"lead\">{data['intro']}</p>"]

    if page_type == "weather":
        parts.append("    <h2>Seasons</h2>\n    <ul class=\"clean\">")
        for s in data["seasons"]:
            parts.append(
                f"      <li><strong>{s['name']} ({s['months']})</strong> — {s['description']}</li>"
            )
        parts.append("    </ul>")
    elif page_type == "attractions":
        parts.append("    <h2>Must-See Attractions</h2>\n    <ul class=\"clean\">")
        for a in data["attractions"]:
            parts.append(f"      <li><strong>{a['name']}</strong> — {a['description']}</li>")
        parts.append("    </ul>")
    elif page_type == "itinerary":
        parts.append("    <h2>Day-by-Day Plan</h2>")
        for d in data["days"]:
            parts.append(
                f"    <h3>{d['day']}</h3>\n    <ul class=\"clean\">\n"
                f"      <li><strong>Morning:</strong> {d['morning']}</li>\n"
                f"      <li><strong>Afternoon:</strong> {d['afternoon']}</li>\n"
                f"      <li><strong>Evening:</strong> {d['evening']}</li>\n"
                f"    </ul>"
            )
    elif page_type == "budget":
        parts.append("    <h2>Daily Costs</h2>\n    <div class=\"table-wrap\">\n      <table>")
        parts.append(
            "        <thead>\n          <tr><th>Category</th><th>Budget</th>"
            "<th>Mid-range</th><th>Luxury</th></tr>\n        </thead>\n        <tbody>"
        )
        for c in data["daily_costs"]:
            parts.append(
                f"          <tr><td>{c['category']}</td><td>{c['budget']}</td>"
                f"<td>{c['mid_range']}</td><td>{c['luxury']}</td></tr>"
            )
        parts.append("        </tbody>\n      </table>\n    </div>")

    parts.append("    <h2>Tips</h2>\n    <ul class=\"clean\">")
    for t in data["tips"]:
        parts.append(f"      <li>{t}</li>")
    parts.append("    </ul>")
    return "\n".join(parts)


def render_page(city: str, page_type: str, data: dict) -> str:
    slug = slugify(city)
    cfg = PAGE_TYPES[page_type]
    title = cfg["title"].format(city=city)
    h1 = cfg["h1"].format(city=city)
    meta = data.get("meta_description", f"{h1}.").replace('"', "&quot;")
    body = render_body(page_type, data)

    related = [c for c in CITIES if c != city][:3]
    related_items = "\n".join(
        f'        <li><a href="{slugify(c)}.html">Best time to visit {c}</a></li>'
        for c in related
    )
    other_pages = "\n".join(
        f'        <li><a href="{slug}-{pt}.html">{PAGE_TYPES[pt]["h1"].format(city=city)}</a></li>'
        for pt in PAGE_TYPES
        if pt != page_type
    )

    return f"""<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>{title}</title>
  <meta name="description" content="{meta}">
  <link rel="canonical" href="{BASE_URL}/cities/{slug}-{page_type}.html">
  <link rel="stylesheet" href="../css/style.css">
</head>
<body>
  <header class="site-header">
    <a class="brand" href="../index.html">Travel Guides</a>
  </header>

  <main>
    <h1>{h1}</h1>
{body}

    <div class="related-links">
      <h2>More on {city}</h2>
      <ul>
        <li><a href="{slug}.html">Best time to visit {city}</a></li>
{other_pages}
      </ul>
    </div>

    <div class="related-links">
      <h2>Related Cities</h2>
      <ul>
{related_items}
      </ul>
    </div>
  </main>

  <footer class="site-footer">
    &copy; 2026 Travel Guides
  </footer>
</body>
</html>
"""

PROMPT_TEMPLATE = """You are a travel writer. Produce factual, concise travel guide content for the city: {city}.

Return ONLY a JSON object (no markdown, no code fences) with exactly this schema:

- Keep descriptions to a single sentence.
- Output valid JSON only.

def call_deepseek(city: str, api_key: str, retries: int = 3) -> dict:
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json",
    }
    payload = {
        "model": API_MODEL,
        "messages": [
            {"role": "system", "content": "You output strict JSON only."},
            {"role": "user", "content": PROMPT_TEMPLATE.format(city=city)},
        ],
        "temperature": 0.7,
        "response_format": {"type": "json_object"},
    }

    last_err = None
    for attempt in range(1, retries + 1):
        try:
            resp = requests.post(API_URL, headers=headers, json=payload, timeout=120)
            if resp.status_code != 200:
                raise RuntimeError(f"HTTP {resp.status_code}: {resp.text[:300]}")
            data = resp.json()
            content = data["choices"][0]["message"]["content"]
            return json.loads(content)
        except Exception as exc:  # noqa: BLE001
            last_err = exc
            print(f"  [warn] {city} 第 {attempt} 次调用失败: {exc}", file=sys.stderr)
            if attempt < retries:
                time.sleep(2 * attempt)
    raise RuntimeError(f"{city} 生成失败: {last_err}")

def rebuild_sitemap() -> None:
    today = date.today().isoformat()
    urls = [
        (f"{SITE_BASE_URL}/index.html", "weekly", "1.0"),
    ]
    for html_file in sorted(CITIES_DIR.glob("*.html")):
        urls.append((f"{SITE_BASE_URL}/cities/{html_file.name}", "monthly", "0.8"))

    entries = "\n".join(
        f"""  <url>
    <loc>{escape(loc)}</loc>
    <lastmod>{today}</lastmod>
    <changefreq>{freq}</changefreq>
    <priority>{prio}</priority>
  </url>"""
        for loc, freq, prio in urls
    )

    xml = f"""<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
{entries}
</urlset>
"""
    SITEMAP_PATH.write_text(xml, encoding="utf-8")
    print(f"[sitemap] 已更新 {SITEMAP_PATH}（{len(urls)} 条 URL）")

    parser.add_argument("--dry-run", action="store_true", help="只打印计划，不调用 API")
    args = parser.parse_args()

    CITIES_DIR.mkdir(parents=True, exist_ok=True)

    targets = CITIES
    if args.only:
        wanted = {c.lower() for c in args.only}
        targets = [c for c in CITIES if c.lower() in wanted]
        missing = wanted - {c.lower() for c in targets}
        if missing:
            print(f"[error] 未知城市: {', '.join(sorted(missing))}", file=sys.stderr)
            return 1

    api_key = os.environ.get(API_KEY_ENV)
    if not api_key and not args.dry_run:
        print(f"[error] 请设置环境变量 {API_KEY_ENV}", file=sys.stderr)
        return 1

    todo = []
    for city in targets:
        out = CITIES_DIR / city_filename(city)
        if out.exists() and not args.force:
            print(f"[skip] {city} 已存在: {out.name}")
            continue
        todo.append(city)

    if not todo:
        print("[info] 没有需要生成的城市。")
        rebuild_sitemap()
        return 0

    print(f"[info] 计划生成 {len(todo)} 个城市: {', '.join(todo)}")
    if args.dry_run:
        return 0

    for i, city in enumerate(todo, 1):

        # 轻微限速，避免触发 API 频率限制
        time.sleep(1)

    rebuild_sitemap()
    return 0


if __name__ == "__main__":
    sys.exit(main())
